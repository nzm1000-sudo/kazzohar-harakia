01 [pointed] וְעוֹנֶה וּמְרַחֵם, בְּכָל עֵת צָרָה וְצוּקָה, אֵין לָנוּ מֶלֶךְ עוֹזֵר וְסוֹמֵךְ
02 [pointed] אֶלָּא אָתָּה:
03 [pointed] אֱלהֵי הָרִאשׁוֹנִים וְהָאַחֲרוֹנִים, אֱלוֹהַּ כָּל בְּרִיּוֹת, אֲדוֹן כָּל תּוֹלָדוֹת,
04 [pointed] הַמְּהֻלָּל בְּכָל הַתִּשְׁבָּחוֹת, הַמְנַהֵג עוֹלָמוֹ בְּחֶסֶד, וּבְרִיּוֹתָיו
05 [pointed] בְּרַחֲמִים. וַיהוָה אֱלהִים אֱמֶת, לא יָנוּם וְלא יִישָׁן, הַמְעוֹרֵר
06 [pointed] יְשֵׁנִים, וְהַמֵּקִיץ נִרְדָּמִים, מְחַיֵּה מֵתִים, וְרוֹפֵא חוֹלִים, פּוֹקֵחַ
07 [pointed] עִוְרִים, וְזוֹקֵף כְּפוּפִים, הַמֵּשִׂיחַ אִלְּמִים, וְהַמְפַעֲנֵחַ נֶעֱלָמִים,
08 [pointed] וּלְךָ לְבַדְּךָ אֲנַחְנוּ מוֹדִים:
09 [pointed] וְאִלּוּ פִּינוּ מָלֵא שִׁירָה כַּיָּם, וּלְשׁוֹנֵנוּ רִנָּה כַּהֲמוֹן גַּלָּיו, וְשִׂפְתוֹתֵינוּ
10 [pointed] שֶׁבַח כְּמֶרְחֲבֵי רָקִיעַ, וְעֵינֵינוּ מְאִירוֹת כַּשֶּׁמֶשׁ וְכַיָּרֵחַ, וְיָדֵינוּ
11 [pointed] פְרוּשׂוֹת כְּנִשְׁרֵי שָׁמַיִם, וְרַגְלֵינוּ קַלּוֹת כָּאַיָּלוֹת, אֵין אֲנַחְנוּ
12 [pointed] מַסְפִּיקִין לְהוֹדוֹת לְךָ יְהוָה אֱלהֵינוּ, וּלְבָרֵךְ אֶת שִׁמְךָ מַלְכֵּנוּ,
13 [pointed] עַל אַחַת מֵאֶלֶף אַלְפֵי אֲלָפִים וְרב רִבֵּי רִבָבוֹת פְּעָמִים, הַטּוֹבוֹת
14 [pointed] נִסִּים וְנִפְלָאוֹת שֶׁעָשִׂיתָ עִמָּנוּ וְעִם אֲבוֹתֵינוּ. מִלְּפָנִים מִמִּצְרַיִם
15 [pointed] גְּאַלְתָּנוּ יְהוָה אֱלהֵינוּ, מִבֵּית עֲבָדִים פְּדִיתָנוּ, בְּרָעָב זַנְתָּנוּ,
16 [pointed] וּבְשָׂבָע כִּלְכַּלְתָּנוּ, מֵחֶרֶב הִצַּלְתָּנוּ, מִדֶּבֶר מִלַּטְתָּנוּ, וּמֵחֳלָאִים
17 [pointed] רָעִים וְרַבִּים דִּלִּיתָנוּ. עַד הֵנָּה עֲזָרוּנוּ רַחֲמֶיךָ, וְלא עֲזָבוּנוּ
18 [pointed] חֲסָדֶיךָ. עַל כֵּן אֵבָרִים שֶׁפִּלַּגְתָּ בָּנוּ, וְרוּחַ וּנְשָׁמָה שֶׁנָּפַחְתָּ בְּאַפֵּינוּ,
19 [pointed] וְלָשׁוֹן אֲשֶׁר שַׂמְתָּ בְּפִינוּ, הֵן הֵם יוֹדוּ, וִיבָרְכוּ, וִישַׁבְּחוּ, וִיפָאֲרוּ
20 [pointed] (וִישׁוֹררוּ), אֶת שִׁמְךָ מַלְכֵּנוּ תָמִיד, כִּי כָל פֶּה לְךָ יוֹדֶה, וְכָל
21 [pointed] לָשׁוֹן לְךָ תְשַׁבֵּחַ, וְכָל עַיִן לְךָ תְצַפֶּה, וְכָל בֶּרֶךְ לְךָ תִכְרַע, וְכָל
22 [pointed] קוֹמָה לְפָנֶיךָ תִשְׁתַּחֲוֶה. וְהַלְּבָבוֹת יִירָאוּךָ, וְהַקֶּרֶב וְהַכְּלָיוֹת
23 [pointed] יְזַמְּרוּ לִשְׁמֶךָ, כַּדָּבָר שֶׁנֶּאֱמַר כָּל עַצְמתַי תּאמַרְנָה יְהוָה מִי
24 [pointed] כָמוֹךָ! מַצִּיל עָנִי מֵחָזָק מִמֶּנּוּ, וְעָנִי ואֶבְיוֹן מִגּזְלוֹ:
25 [pointed] שַׁוְעַת עֲנִיִּים אַתָּה תִּשְׁמַע, צַעֲקַת הַדַּל תַּקְשִׁיב וְתוֹשִׁיעַ, וכָתוּב
26 [pointed] רַנְּנוּ צַדִּיקִים בַּיהוָה, לַיְשָׁרִים נָאוָה תְהִלָּה:
