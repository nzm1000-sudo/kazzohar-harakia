01 [pointed] יָ פִין לָהּ אֶלָּא שֶׁאֵין מַכְנִיסִין מֵי רַגְלַיִם בַּמִּקְדָּשׁ מִפְּנֵי הַכָּבוֹד:
02 [pointed] תַּנְיָא, רִבִּי נָתָן אוֹמֵר: כְּשֶׁהוּא שׁוֹחֵק אוֹמֵר הָדֵק הֵיטֵב. הֵיטֵב
03 [pointed] הָדֵק. מִפְּנֵי שֶׁהַקּוֹל יָפֶה לַבְּשָׂמִים. פִּטְּמָהּ לַחֲצָאִין כְּשֵׁרָה. לְשָׁלִישׁ
04 [pointed] וּלְרָבִיעַ לא שָׁמַעְנוּ. אָמַר רִבִּי יְהוּדָה: זֶה הַכְּלָל: אִם כְּמִדָּתָהּ
05 [pointed] כְּ שֵׁרָה לַחֲצָאִין. וְאִם חִסֵּר אַחַת מִכָּל-סַמֲמָנֶיהָ חַיָּב מִיתָה:
06 [pointed] תָּנֵי בַר קַפָּרָא: אַחַת לְשִׁשִּׁים אוֹ לְשִׁבְעִים שָׁנָה הָיְתָה בָאָה שֶׁל
07 [pointed] שִׁירַיִם לַחֲצָאִין. וְעוֹד תָּנֵי בַר קַפָּרָא: אִלּוּ הָיָה נוֹתֵן בָּהּ קָרְטוֹב
08 [pointed] שֶׁל דְּבַשׁ אֵין אָדָם יָכוֹל לַעֲמוֹד מִפְּנֵי רֵיחָהּ. וְלָמָּה אֵין מְעָרְבִין
09 [pointed] בָּהּ דְּבַשׁ מִפְּנֵי שֶׁהַתּוֹרָה אָמְרָה: כִּי כָל-שְׂאר וְכָל-דְּבַשׁ לא-
10 [pointed] תַקְטִירוּ מִמֶּנּוּ אִשֶּׁה לַה’: ה’ צְבָאוֹת עִמָּנוּ. מִשְׂגָּב לָנוּ אֱלהֵי
11 [pointed] יַעֲקב סֶלָה: ה’ צְבָאוֹת. אַשְׁרֵי אָדָם בּטֵחַ בָּךְ: ה’ הוֹשִׁיעָה. הַמֶּלֶךְ
12 [pointed] יַעֲנֵנוּ בְיוֹם-קָרְאֵנוּ: וְעָרְבָה לַה’ מִנְחַת יְהוּדָה וִירוּשָׁלָיִם. כִּימֵי
13 [pointed] עוֹלָם וּכְשָׁנִים קַדְמנִיּוֹת
14 [pointed] ויאמר משה אל אהרן קח את המחתה ותן עליה אש מעל
15 [pointed] המזבח ושים קטורת והולך מהרה אל העדה וכפר עליהם, כי
16 [pointed] יצא הקצף מלפני ה’ החל הנגף. ויקח אהרן כאשר דבר משה
17 [pointed] וירץ אל תוך הקהל והנה החל הנגף בעם, ויתן את הקטורת
18 [pointed] ויכפר על העם.
