01 [pointed] אֲשֶׁר-יִלְקְטוּ יוֹם יוֹם: וַיּאמֶר משֶׁה וְאַהֲרן אֶל-כָּל-בְּנֵי יִשְׂרָאֵל,
02 [pointed] עֶרֶב וִידַעְתֶּם כִּי ה’ הוֹצִיא אֶתְכֶם מֵאֶרֶץ מִצְרָיִם: וּבקֶר וּרְאִיתֶם
03 [pointed] אֶת-כְּבוֹד ה’ בְּשָׁמְעוֹ אֶת-תְּלֻנּתֵיכֶם עַל-ה’, וְנַחְנוּ מָה כִּי תַלִּינוּ
04 [pointed] עָלֵינוּ: וַיאמֶר משֶׁה בְּתֵת ה’ לָכֶם בָּעֶרֶב בָּשָׂר לֶאֱכל וְלֶחֶם בַּבּקֶר
05 [pointed] לִשְׂבּעַ בִּשְׁמעַ ה’ אֶת-תְּלֻנּתֵיכֶם אֲשֶׁר-אַתֶּם מַלִּינִם עָלָיו, וְנַחְנוּ
06 [pointed] מָה לא-עָלֵינוּ תְלֻנּתֵיכֶם כִּי עַל-ה:’
07 [pointed] וַיּאמֶר משֶׁה אֶל-אַהֲרן אֱמר אֶל-כָּל-עֲדַת בְּנֵי יִשְׂרָאֵל קִרְבוּ
08 [pointed] לִפְנֵי ה’, כִּי שָׁמַע אֵת תְּלֻנתֵיכֶם: וַיְהִי כְּדַבֵּר אַהֲרן אֶל-כָּל-עֲדַת
09 [pointed] בְּנֵי-יִשְׂרָאֵל וַיִּפְנוּ אֶל-הַמִּדְבָּר, וְהִנֵּה כְּבוֹד ה’ נִרְאָה בֶּעָנָן: וַיְדַבֵּר
10 [pointed] ה’ אֶל-משֶׁה לֵּאמר: שָׁמַעְתִּי אֶת תְּלוּנת בְּנֵי יִשְׂרָאֵל דַּבֵּר אֲלֵהֶם
11 [pointed] לֵאמר בֵּין הָעַרְבַּיִם תּאכְלוּ בָשָׂר וּבַבּקֶר תִּשְׂבְּעוּ-לָחֶם, וִידַעְתֶּם
12 [pointed] כִּי אֲנִי ה’ אֱלהֵיכֶם: וַיְהִי בָעֶרֶב וַתַּעַל הַשְּׂלָו וַתְּכַס אֶת-הַמַּחֲנֶה
13 [pointed] וּבַבּקֶר הָיְתָה שִׁכְבַת הַטָּל סָבִיב לַמַּחֲנֶה: וַתַּעַל שִׁכְבַת הַטָּל,
14 [pointed] וְהִנֵּה עַל-פְּנֵי הַמִּדְבָּר דַּק מְחֻסְפָּס דַּק כַּכְּפר עַל-הָאָרֶץ: וַיִּרְאוּ
15 [pointed] בְנֵי-יִשְׂרָאֵל וַיּאמְרוּ אִישׁ אֶל-אָחִיו מָן הוּא כִּי לא יָדְעוּ מַה-הוּא,
16 [pointed] וַיּאמֶר משֶׁה אֲלֵהֶם הוּא הַלֶּחֶם אֲשֶׁר נָתַן ה’ לָכֶם לְאָכְלָה: זֶה
17 [pointed] הַדָּבָר אֲשֶׁר צִוָּה ה’ לִקְטוּ מִמֶּנּוּ אִישׁ לְפִי אָכְלוֹ, עמֶר לַגֻּלְגלֶת
18 [pointed] מִסְפַּר נַפְשׁתֵיכֶם אִישׁ לַאֲשֶׁר בְּאָהֳלוֹ תִּקָּחוּ: וַיַּעֲשׂוּ-כֵן בְּנֵי
19 [pointed] יִשְׂרָאֵל, וַיִּלְקְטוּ הַמַּרְבֶּה וְהַמַּמְעִיט: וַיָּמדּוּ בָעמֶר וְלא הֶעְדִּיף
20 [pointed] הַמַּרְבֶּה וְהַמַּמְעִיט לא הֶחְסִיר, אִישׁ לְפִי-אָכְלוֹ לָקָטוּ:
21 [pointed] וַיּאמֶר משֶׁה אֲלֵהֶם, אִישׁ אַל-יוֹתֵר מִמֶּנּוּ עַד-בּקֶר: וְלא-שָׁמְעוּ
22 [pointed] אֶל-משֶׁה וַיּוֹתִרוּ אֲנָשִׁים מִמֶּנּוּ עַד-בּקֶר וַיָּרֻם תּוֹלָעִים וַיִּבְאַשׁ,
23 [pointed] וַיִּקְצף עֲלֵהֶם משֶׁה: וַיִּלְקְטוּ אתוֹ בַּבקֶר בַּבּקֶר אִישׁ כְּפִי אָכְלוֹ,
24 [pointed] וְחַם הַשֶּׁמֶשׁ וְנָמָס: וַיְהִי בַּיּוֹם הַשִּׁשִּׁי לָקְטוּ לֶחֶם מִשְׁנֶה שְׁנֵי
25 [pointed] הָעמֶר לָאֶחָד, וַיָבאוּ כָּל-נְשִׂיאֵי הָעֵדָה וַיַּגִּידוּ לְמשֶׁה: וַיאמֶר
26 [pointed] אֲלֵהֶם הוּא אֲשֶׁר דִּבֶּר ה’ שַׁבָּתוֹן שַׁבַּת-קדֶשׁ לַה’ מָחָר, אֵת
27 [pointed] אֲשֶׁר-תּאפוּ אֵפוּ וְאֵת אֲשֶׁר-תְּבַשְּׁלוּ בַּשֵּׁלוּ וְאֵת כָּל-הָעדֵף הַנִּיחוּ
