01 [pointed] יְהִי כְבוֹד יְיָ לְעוֹלָם יִשְמַח יְיָ בְּמַעֲשָיו: יְהִי שֵׁם יְיָ מְבוֹרָךְ מֵעַתָּה
02 [pointed] וְעַד עוֹלָם:
03 [pointed] הַמַּבִּיט לָאָרֶץ וַתִּרְעָד יִגַּע בֶּהָרִים וְיֶעֱשָׁנוּ: רוּמָה עַל הַשָּׁמַיִם
04 [pointed] אֱלהִים עַל כָּל הָאָרֶץ כְּבוֹדֶךָ:
05 [heading] שלשה עשר עיקרים
06 [prose] קטע אמוני זה, מחזק אצל כל יהודי את החיבור הרוחני לקב”ה
07 [prose] ומבליט את הידיעה וההבנה, שיש בורא לעולם, והעולם אינו הפקר
08 [prose] חלילה.
09 [prose] התמדה בקריאת קטע זה על בסיס יומיומי תביא את האדם
10 [prose] למודעות גבוהה ולביטחון עצום ומכך יבין שכל דבר נעשה בהוראתו
11 [prose] של הקב”ה ומסתמא, שיוקל לו לאדם.
12 [pointed] הֲרֵינִי מַאֲמִין בֶּאֱמוּנָה שְׁלֵמָה בִּשְׁלשָׁה עָשָׂר עִקָּרִים שֶׁל הַתּוֹרָה
13 [pointed] הַקְּדוֹשָׁה: (א) שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא וּמַשְׁגִּיחַ, (ב) וְהוּא
14 [pointed] אֶחָד, (ג) וְאֵין לוֹ גּוּף וְאֵין לוֹ דְּמוּת הַגּוּף, (ד) וְהוּא קַדְמוֹן
15 [pointed] לְכָל קְדוּמִים, (ה) וְאֵין עֲבוֹדָה לְזוּלָתוֹ, (ו) וְיוֹדֵעַ מַחְשְׁבוֹת בְּנֵי
16 [pointed] אָדָם, (ז) וְשֶהַנְּבוּאָה אֱמֶת, (ח) וְשֶׁמּשֶׁה רַבֵּינוּ עָלָיו הַשָּׁלוֹם
17 [pointed] הוּא אֲדוֹן לְכָל הַנְּבִיאִים, (ט) וְשֶׁהַתּוֹרָה כֻלָּהּ נְתוּנָה מִן הַשָּׁמַיִם,
18 [pointed] (י) וְשֶׁלּא תִּשְׁתַּנֶּה בְּשׁוּם זְמַן חַס וְשָׁלוֹם, (יא) וְשֶׁהַקָּדוֹשׁ בָּרוּךְ
19 [pointed] הוּא מַעֲנִישׁ לָרְשָׁעִים וּמְשַׁלֵּם שָׂכָר טוֹב לַצַּדִּיקִים, (יב) וְשֶׁיָּבוֹא
20 [pointed] מֶלֶךְ הַמָּשִׁיחַ, (יג) וְשֶׁהַמֵּתִים עֲתִידִים לִחְיוֹת, יֶהִי רָצוֹן מִלְּפָנֶיךָ
21 [pointed] יְהוָה אֱלהֵינוּ וֵאלהֵי אֲבתֵינוּ שֶׁתָּכף יִצְרֵנוּ לַעֲבוֹדָתֶךָ כָּל יְמֵי
22 [pointed] חַיֵּינוּ תָּמִיד, אָמֵן כֵּן יְהִי רָצוֹן:
