01 [bold] צָרָתִי אֲדֹנָי דָּרָשְׁתִּי יָדִי לַיְלָה נִגְּרָה וְלֹא תָפוּג מֵאֲנָה הִנָּחֵם
02 [bold] נַפְשִׁי: {ד} אֶזְכְּרָה אֱלֹהִים וְאֶהֱמָיָה אָשִׂיחָה וְתִתְעַטֵּף
03 [bold] רוּחִי סֶלָה: {ה} אָחַזְתָּ שְׁמֻרוֹת עֵינָי נִפְעַמְתִּי וְלֹא אֲדַבֵּר:
04 [bold] {ו} חִשַּׁבְתִּי יָמִים מִקֶּדֶם שְׁנוֹת עוֹלָמִים: {ז} אֶזְכְּרָה נְגִינָתִי
05 [bold] בַּלָּיְלָה עִם לְבָבִי אָשִׂיחָה וַיְחַפֵּשׂ רוּחִי: {ח} הַלְעוֹלָמִים
06 [bold] יִזְנַח אֲדֹנָי וְלֹא יֹסִיף לִרְצוֹת עוֹד: {ט} הֶאָפֵס לָנֶצַח
07 [bold] חַסְדּוֹ גָּמַר אֹמֶר לְדֹר וָדֹר: {י} הֲשָׁכַח חַנּוֹת אֵל אִם קָפַץ
08 [bold] בְּאַף רַחֲמָיו סֶלָה: {יא} וָאֹמַר חַלּוֹתִי הִיא שְׁנוֹת יְמִין
09 [bold] עֶלְיוֹן: {יב} (אזכיר) אֶזְכּוֹר מַעַלְלֵי יָהּ כִּי אֶזְכְּרָה מִקֶּדֶם
10 [bold] פִּלְאֶךָ: {יג} וְהָגִיתִי בְכָל פָּעֳלֶךָ וּבַעֲלִילוֹתֶיךָ אָשִׂיחָה:
11 [bold] {יד} אֱלֹהִים בַּקֹּדֶשׁ דַּרְכֶּךָ מִי אֵל גָּדוֹל כֵּאלֹהִים: {טו} אַתָּה
12 [bold] הָאֵל עֹשֵׂה פֶלֶא הוֹדַעְתָּ בָעַמִּים עֻזֶּךָ: {טז} גָּאַלְתָּ בִּזְרוֹעַ
13 [bold] עַמֶּךָ בְּנֵי יַעֲקֹב וְיוֹסֵף סֶלָה: {יז} רָאוּךָ מַּיִם אֱלֹהִים
14 [bold] רָאוּךָ מַּיִם יָחִילוּ אַף יִרְגְּזוּ תְהֹמוֹת: {יח} זֹרְמוּ מַיִם
15 [bold] עָבוֹת קוֹל נָתְנוּ שְׁחָקִים אַף חֲצָצֶיךָ יִתְהַלָּכוּ: {יט} קוֹל
16 [bold] רַעַמְךָ בַּגַּלְגַּל הֵאִירוּ בְרָקִים תֵּבֵל רָגְזָה וַתִּרְעַשׁ הָאָרֶץ:
17 [bold] {כ} בַּיָּם דַּרְכֶּךָ (ושביליך) וּשְׁבִילְךָ בְּמַיִם רַבִּים וְעִקְּבוֹתֶיךָ
18 [bold] לֹ א נֹדָעוּ: {כא} נָחִיתָ כַצֹּאן עַמֶּךָ בְּיַד מֹשֶׁה וְאַהֲרֹן:
19 [bold] תהילים פרק-צ
20 [bold] {א} תְּפִלָּה לְמֹשֶׁה אִישׁ הָאֱלֹהִים אֲדֹנָי מָעוֹן אַתָּה הָיִיתָ
21 [bold] לָּנוּ בְּדֹר וָדֹר: {ב} בְּטֶרֶם הָרִים יֻלָּדוּ וַתְּחוֹלֵל אֶרֶץ וְתֵבֵל
22 [bold] וּמֵעוֹלָם עַד עוֹלָם אַתָּה אֵל: {ג} תָּשֵׁב אֱנוֹשׁ עַד דַּכָּא
