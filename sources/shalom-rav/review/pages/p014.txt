01 [bold] וְעָלֵינוּ, יֶהֱמוּ נָא רַחֲמֶיךָ יְהֹוָה אֱלֹהֵינוּ, וְתֵן שָׂכָר טוֹב לְכָל
02 [bold] הַבּוֹטְחִים בְּשִׁמְךָ בֶאֱמֶת, וְשִׂים חֶלְקֵנוּ עִמָּהֶם, וּלְעוֹלָם לֹא
03 [bold] נֵבוֹשׁ כִּי בְךָ בָטַחְנוּ, וְעַל חַסְדְּךָ הַגָּדוֹל בֶּאֱמֶת נִשְׁעָנְנוּ.
04 [bold] בָּרוּךְ אַתָּה יוּהוּווּהוּ, מִשְׁעָן וּמִבְטָח לַצַּדִּיקִים:
05 [bold] תִּשְׁכּוֹן בְּתוֹךְ יְרוּשָׁלַיִם עִירְךָ כַּאֲשֶׁר דִּבַּרְתָּ, וְכִסֵּא דָוִד
06 [bold] עַבְדְּךָ, מְהֵרָה בְתוֹכָהּ תָּכִין, וּבְנֵה אוֹתָהּ בִּנְיַן עוֹלָם בִּמְהֵרָה
07 [bold] בְיָמֵינוּ:
08 [bold] (בתשעה באב מוסיפים:) נַחֵם יְהֹוָה אֱלֹהֵינוּ אֶת אֲבֵלֵי צִיּוֹן וְאֶת
09 [bold] אֲבֵלֵי יְרוּשָׁלַיִם, וְאֶת הָעִיר הַחֲרֵבָה וְהַבְּזוּיָה וְהַשוֹמֵמָה. מִבְּלִי
10 [bold] בָנֶיהָ הִיא יוֹשֶׁבֶת וְרֹאשָׁהּ חָפוּי כְּאִשָּׁה עֲקָרָה שֶׁלֹּא יָלָדָה. וַיְבַלְּעוּהָ
11 [bold] לִיגְיוֹנִים וְיִירָשׁוּה, וַיַטִּילוֹ אֶת עַמְּךָ יִשְׂרָאֵל לַחֶרֶב, וַיַּהַרְגוּ בְּזָדוֹן
12 [bold] חֲסִידָי עֶלְיוֹן. עַל כֵּן צִיּוֹן בְּמֶרֶר תִבְכֶּה וִירוּשָׁלָיִם תִּתֵּן קוֹלָהּ. לִבִּי
13 [bold] לִבִּי עַל חַלְלֵיהֶם, מֵעַי מֵעַי עַל הֲרוּגֵיהֶם, כִּי אַתָּה יְהֹוָה בָּאֵשׁ הַצָּתָהּ
14 [bold] וּבָאֵשׁ אַתָּה עָתִיד לִבְנוֹתָהּ. כַּכָּתוּב: וַאֲנִי אֶהְיֶה לָהּ נְאֻם יְהֹוָה חוֹמַת
15 [bold] אֵשׁ סָבִיב וּלְכָבוֹד אֶהְיֶה בְּתוֹכָהּ.
16 [bold] בָּרוּךְ אַתָּה יְהֹוָה, מְנַחֵם צִיּוֹן בְּבִנְיַן יְרוּשָׁלַיִם. (וממשיכים “את
17 [bold] צמח” וכו’..)
18 [bold] בָּרוּךְ אַתָּה יֻהֻוֻהֻ, בּוֹנֵה יְרוּשָׁלָיִם:
19 [bold] אֶת צֶמַח דָּוִד עַבְדְּךָ מְהֵרָה תַצְמִיחַ, וְקַרְנוֹ תָרוּם בִּישׁוּעָתֶךָ.
20 [bold] כִּי לִישׁוּעָתְךָ קִוִּינוּ וְצִפִּינוּ כָּל הַיּוֹם. בָּרוּךְ אַתָּה יִהִוִהִ,
21 [bold] מַצְמִיחַ קֶרֶן יְשׁוּעָה:
