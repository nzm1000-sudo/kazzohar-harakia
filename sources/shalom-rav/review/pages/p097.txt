01 [pointed] בֵּין בְּאֹנֶס בֵּין בְּרָצוֹן בֵּין בְּכַוָּנָה בֵּין שֶׁלֹּא בְכַוָּנָה, בְּכָל־
02 [pointed] לָשׁוֹן, בֵּין בִּלְשׁוֹן נְדָרִים וּשְׁבוּעוֹת אוֹ נְזִירוֹת אוֹ אִסָּרִים
03 [pointed] אוֹ קוּנָמוֹת אוֹ חֲרָמוֹת אוֹ הַסְכָּמוֹת אוֹ קַבָּלָה בַלֵּב,
04 [pointed] בְּכָל־צַד וְאֹפֶן שֶׁיִּהְיֶה - הֲרֵי אֲנַחְנוּ מְבַטְּלִים אוֹתָם בִּטּוּל
05 [pointed] גָּמוּר, וּמַתְנִים עֲלֵיהֶם שֶׁיִּהְיוּ כֻלָּם בְּטֵלִים וּמְבֻטָּלִים,
06 [pointed] לָא שְׁרִירִין וְלָא קַיָּמִין, וְלֹא יִהְיוּ חָלִין כְּלָל. וַהֲרֵי אֲנַחְנוּ
07 [pointed] מֵעַתָּה מְגַלִּים דַּעְתֵּנוּ שֶׁאֵין לָנוּ חֵפֶץ בָּהֶם וְלֹא בְקִיּוּמָם,
08 [pointed] כִּי אִם בְּקִיּוּם תְּנַאי מוֹדָעָה זוֹ, שֶׁלֹּא תִתְבַּטֵּל לְעוֹלָם.
09 [pointed] וְעוֹד אֲנַחְנוּ מוֹסְרִים מוֹדָעָה לִפְנֵיכֶם בְּכָל־לִבֵּנוּ וּבְכָל־
10 [pointed] נַפְשֵׁנוּ, וַהֲרֵי אֲנַחְנוּ מְבַטְּלִים מֵעַתָּה וְעַד עוֹלָם כָּל־מִין
11 [pointed] מַעֲשֶׂה שֶׁל אִסּוּר אִם נַעֲשֶׂה חַס וְשָׁלוֹם, וְכֵן כָּל־מִין
12 [pointed] דִּבּוּר שֶׁל אִסּוּר אִם יֵצֵא מִפִּינוּ חַס וְשָׁלוֹם, מֵהַמַּעֲשִׂים
13 [pointed] וּמֵהַדִּבּוּרִים הָאֲסוּרִים מִדְּאוֹרַיְתָא אוֹ מִדְּרַבָּנָן, אוֹ מִדִּבְרֵי
14 [pointed] קַבָּלָה אוֹ מִדִּבְרֵי סוֹפְרִים. וְגַם כָּל־מִינֵי מַחֲשָׁבוֹת זָרוֹת
15 [pointed] וְהִרְהוּרִים רָעִים, וְגַם כָּל־מִינֵי גַאֲוָה וְכַעַס וְקַפְדָּנוּת,
16 [pointed] וְקִנְאָה וְשִׂנְאָה וְתַחֲרוּת, וְחֶמְדָּה וְתַאֲוָה וּנְקִימָה וּנְטִירָה,
17 [pointed] וְכָל־מִינֵי רְאִיָּה וּשְׁמִיעָה וְרֵיחַ וְקוֹל וּרְמִיזָה הָאֲסוּרִים,
18 [pointed] אִם נַעֲשֶׂה אוֹ נְדַבֵּר אוֹ נַחֲשֹׁב אוֹתָם חַס וְשָׁלוֹם, בֵּין
19 [pointed] בְּאֹנֶס בֵּין בְּרָצוֹן, בֵּין בְּכַעַס, בֵּין בְּשׁוֹגֵג בֵּין בְּמֵזִיד, בֵּין
20 [pointed] בְּכַוָּנָה בֵּין שֶׁלֹּא בְכַוָּנָה, בֵּין בַּחֲלוֹם בֵּין בְּהָקִיץ. מֵעַתָּה
21 [pointed] הֲרֵי אֲנַחְנוּ מִתְחָֽרְטִים עֲלֵיהֶם, וּמַתְנִים עֲלֵיהֶם שֶׁיִּהְיוּ
22 [pointed] כֻלָּם בְּטֵלִים וּמְבֻטָּלִים מֵעִקָּרָן, וְלֹא יִהְיוּ חָלִין כְּלָל
23 [pointed] וְעִקָּר, וּבְכֻלְּהוֹן אִתְחָרַטְנָא בְהוֹן מֵעַתָּה וְעַד עוֹלָם.
