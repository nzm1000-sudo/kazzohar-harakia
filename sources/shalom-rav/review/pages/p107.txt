01 [bold] {ז} הוּא יְהוָה אֱלֹהֵינוּ בְּכָל הָאָרֶץ מִשְׁפָּטָיו: {ח} זָכַר
02 [bold] לְעוֹלָם בְּרִיתוֹ דָּבָר צִוָּה לְאֶלֶף דּוֹר: {ט} אֲשֶׁר כָּרַת אֶת
03 [bold] אַבְרָהָם וּשְׁבוּעָתוֹ לְיִשְׂחָק: {י} וַיַּעֲמִידֶהָ לְיַעֲקֹב לְחֹק
04 [bold] לְיִשְׂרָאֵל בְּרִית עוֹלָם: {יא} לֵאמֹר לְךָ אֶתֵּן אֶת אֶרֶץ כְּנָעַן
05 [bold] חֶבֶל נַחֲלַתְכֶם: {יב} בִּהְיוֹתָם מְתֵי מִסְפָּר כִּמְעַט וְגָרִים בָּהּ:
06 [bold] {יג} וַיִּתְהַלְּכוּ מִגּוֹי אֶל גּוֹי מִמַּמְלָכָה אֶל עַם אַחֵר: {יד} לֹא
07 [bold] הִנִּיחַ אָדָם לְעָשְׁקָם וַיּוֹכַח עֲלֵיהֶם מְלָכִים: {טו} אַל תִּגְּעוּ
08 [bold] בִמְשִׁיחָי וְלִנְבִיאַי אַל תָּרֵעוּ: {טז} וַיִּקְרָא רָעָב עַל הָאָרֶץ
09 [bold] כָּל מַטֵּה לֶחֶם שָׁבָר: {יז} שָׁלַח לִפְנֵיהֶם אִישׁ לְעֶבֶד נִמְכַּר
10 [bold] יוֹסֵף: {יח} עִנּוּ בַכֶּבֶל (רגליו) רַגְלוֹ בַּרְזֶל בָּאָה נַפְשׁוֹ:
11 [bold] {יט} עַד עֵת בֹּא דְבָרוֹ אִמְרַת יְהוָה צְרָפָתְהוּ: {כ} שָׁלַח
12 [bold] מֶלֶךְ וַיַּתִּירֵהוּ מֹשֵׁל עַמִּים וַיְפַתְּחֵהוּ: {כא} שָׂמוֹ אָדוֹן
13 [bold] לְבֵיתוֹ וּמֹשֵׁל בְּכָל קִנְיָנוֹ: {כב} לֶאְסֹר שָׂרָיו בְּנַפְשׁוֹ וּזְקֵנָיו
14 [bold] יְחַכֵּם: {כג} וַיָּבֹא יִשְׂרָאֵל מִצְרָיִם וְיַעֲקֹב גָּר בְּאֶרֶץ חָם:
15 [bold] {כד} וַיֶּפֶר אֶת עַמּוֹ מְאֹד וַיַּעֲצִמֵהוּ מִצָּרָיו: {כה} הָפַךְ
16 [bold] לִבָּם לִשְׂנֹא עַמּוֹ לְהִתְנַכֵּל בַּעֲבָדָיו: {כו} שָׁלַח מֹשֶׁה עַבְדּוֹ
17 [bold] אַהֲרֹן אֲשֶׁר בָּחַר בּוֹ: {כז} שָׂמוּ בָם דִּבְרֵי אֹתוֹתָיו וּמֹפְתִים
18 [bold] בְּאֶרֶץ חָם: {כח} שָׁלַח חֹשֶׁךְ וַיַּחְשִׁךְ וְלֹא מָרוּ אֶת (דברוו)
19 [bold] דְּבָרוֹ: {כט} הָפַךְ אֶת מֵימֵיהֶם לְדָם וַיָּמֶת אֶת דְּגָתָם:
20 [bold] {ל} שָׁרַץ אַרְצָם צְפַרְדְּעִים בְּחַדְרֵי מַלְכֵיהֶם: {לא} אָמַר
21 [bold] וַיָּבֹא עָרֹב כִּנִּים בְּכָל גְּבוּלָם: {לב} נָתַן גִּשְׁמֵיהֶם בָּרָד אֵשׁ
22 [bold] לֶהָבוֹת בְּאַרְצָם: {לג} וַיַּךְ גַּפְנָם וּתְאֵנָתָם וַיְשַׁבֵּר עֵץ גְּבוּלָם:
23 [bold] {לד} אָמַר וַיָּבֹא אַרְבֶּה וְיֶלֶק וְאֵין מִסְפָּר: {לה} וַיֹּאכַל
