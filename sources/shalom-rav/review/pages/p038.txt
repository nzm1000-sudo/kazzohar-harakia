01 [pointed] לָכֶם לְמִשְׁמֶרֶת עַד-הַבּקֶר: וַיַּנִּיחוּ אתוֹ עַד-הַבּקֶר כַּאֲשֶׁר צִוָּה
02 [pointed] משֶׁה, וְלא הִבְאִישׁ וְרִמָּה לא-הָיְתָה-בּוֹ:
03 [pointed] וַיּאמֶר משֶׁה אִכְלֻהוּ הַיּוֹם כִּי-שַׁבָּת הַיּוֹם לַה’, הַיּוֹם לא תִמְצָאֻהוּ
04 [pointed] בַּשָּׂדֶה: שֵׁשֶׁת יָמִים תִּלְקְטֻהוּ, וּבַיּוֹם הַשְּׁבִיעִי שַׁבָּת לא יִהְיֶה-בּוֹ:
05 [pointed] וַיְהִי בַּיּוֹם הַשְּׁבִיעִי יָצְאוּ מִן-הָעָם לִלְקט וְלא מָצָאו:
06 [pointed] וַיּאמֶר ה’ אֶל-משֶׁה, עַד-אָנָה מֵאַנְתֶּם לִשְׁמר מִצְוֹתַי וְתוֹרתָי: רְאוּ
07 [pointed] כִּי-ה’ נָתַן לָכֶם הַשַּׁבָּת עַל-כֵּן הוּא נתֵן לָכֶם בַּיּוֹם הַשִּׁשִּׁי לֶחֶם
08 [pointed] יוֹמָיִם, שְׁבוּ אִישׁ תַּחְתָּיו אַל-יֵצֵא אִישׁ מִמְּקמוֹ בַּיּוֹם הַשְּׁבִיעִי:
09 [pointed] וַיִּשְׁבְּתוּ הָעָם בַּיּוֹם הַשְּׁבִעִי: וַיִּקְרְאוּ בֵית-יִשְׂרָאֵל אֶת-שְׁמוֹ מָן,
10 [pointed] וְהוּא כְּזֶרַע גַּד לָבָן וְטַעְמוֹ כְּצַפִּיחִת בִּדְבָשׁ: וַיּאמֶר משֶׁה זֶה
11 [pointed] הַדָּבָר אֲשֶׁר צִוָּה ה’ מְלא הָעמֶר מִמֶּנּוּ לְמִשְׁמֶרֶת לְדרתֵיכֶם, לְמַעַן
12 [pointed] יִרְאוּ אֶת-הַלֶּחֶם אֲשֶׁר הֶאֱכַלְתִּי אֶתְכֶם בַּמִּדְבָּר בְּהוֹצִיאִי אֶתְכֶם
13 [pointed] מֵאֶרֶץ מִצְרָיִם: וַיּאמֶר משֶׁה-אֶל אַהֲרן קַח צִנְצֶנֶת אַחַת וְתֶן-
14 [pointed] שָׁמָּה מְלא-הָעמֶר מָן, וְהַנַּח אתוֹ לִפְנֵי ה’ לְמִשְׁמֶרֶת לְדרתֵיכֶם:
15 [pointed] כַּאֲשֶׁר צִוָּה ה’ אֶל-משֶׁה, וַיַּנִּיחֵהוּ אַהֲרן לִפְנֵי הָעֵדֻת לְמִשְׁמָרֶת:
16 [pointed] וּבְנֵי יִשְׂרָאֵל אָכְלוּ אֶת-הַמָּן אַרְבָּעִים שָׁנָה עַד-בּאָם אֶל-אֶרֶץ
17 [pointed] נוֹשָׁבֶת, אֶת-הַמָּן אָכְלוּ עַד-בּאָם אֶל-קְצֵה אֶרֶץ כְּנָעַן: וְהָעמֶר
18 [pointed] עֲשִׂרִית הָאֵיפָה הוּא:
