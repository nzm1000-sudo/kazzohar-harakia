01 [pointed] וְעוֹד אֲנַחְנוּ מוֹסְרִים מוֹדָעָה לִפְנֵיכֶם, רַבּוֹתֵינוּ, שֶׁכָּל־
02 [pointed] מַחֲשָׁבָה רָעָה וְהִרְהוּר רַע וּרְעוּתָא דְלִבָּא, אִם יַעֲלֶה
03 [pointed] בְלִבֵּנוּ וּמֹחֵנוּ, וּבִפְרָט בְּעֵת הַתְּפִלָּה וְתַלְמוּד תּוֹרָה,
04 [pointed] בְּאֵיזֶה צַד וְאֹפֶן שֶׁאֵינוֹ לִכְבוֹד וּרְצוֹן הַבּוֹרֵא יִתְבָּרַךְ,
05 [pointed] מֵעַכְשָׁיו אֲנַחְנוּ מְבַטְּלִים אוֹתָם מַחֲשָׁבוֹת וְהִרְהוּרִים
06 [pointed] רָעִים וּרְעוּתָא דְלִבָּא שֶׁהֵם נֶגֶד רְצוֹן קֻדְשָׁא בְרִיךְ הוּא
07 [pointed] וּשְׁכִינְתֵּהּ, בִּטּוּל גָּמוּר כַּחֶרֶס הַנִּשְׁבָּר. וּמִכָּל־שֶׁכֵּן אִם
08 [pointed] נְדַבֵּר אֵיזֶה דִבּוּר רַע וְאָסוּר, מֵעַכְשָׁיו יִהְיוּ הַכֹּל בְּטֵלִים
09 [pointed] וּמְבֻטָּלִים. וְעַתָּה אֲנַחְנוּ מְבָאֲרִים וּמְפָֽרְשִׁים בַּאֵר הֵיטֵב,
10 [pointed] וַאֲנַחְנוּ מְגַלִּים דַּעְתֵּנוּ וּרְצוֹנֵנוּ בְּכָל־לִבֵּנוּ, כִּי רְצוֹנֵנוּ
11 [pointed] וּמַאֲוַיֵּנוּ וְכַוָּנָתֵנוּ לַעֲבֹד אֶת בּוֹרְאֵנוּ, אֱלֹהֵי אַבְרָהָם אֱלֹהֵי
12 [pointed] יִצְחָק וְיִשְׂרָאֵל, עֲבוֹדָה שְׁלֵמָה, עֲבוֹדָה תַמָּה, בְּמַחֲשָׁבָה
13 [pointed] וְדִבּוּר וּמַעֲשֶׂה, בְּיִרְאָה וְאַהֲבָה וְשִׂמְחָה רַבָּה כְּדָת
14 [pointed] מַה־לַּעֲשׂוֹת. וְכָל־ מַחֲשָׁבָה וְדִבּוּר וּמַעֲשֶׂה שֶׁהֵם נֶגֶד
15 [pointed] רְצוֹנוֹ יִתְבָּרַךְ, מֵעַכְשָׁיו יִהְיוּ בְטֵלִים וּמְבֻטָּלִים, כִּי הֵם
16 [pointed] מִצַּד יֵצֶר הָרַע וְהַכֹּל הָבֶל. בְּאֹפֶן שֶׁכָּל־עֲבוֹדָתֵנוּ לַבּוֹרֵא
17 [pointed] יִתְבָּרַךְ בְּמַחֲשָׁבָה וְדִבּוּר וּמַעֲשֶׂה, הַכֹּל הוּא לַעֲשׂוֹת
18 [pointed] נַחַת רוּחַ לְפָנָיו דַּוְקָא בְּלִי שׁוּם פְּנִיָּה כְּלָל וְעִקָּר. וְהַכֹּל
19 [pointed] הוּא לְשֵׁם יִחוּד קֻדְשָׁא בְרִיךְ הוּא וּשְׁכִינְתֵּהּ, בִּדְחִילוּ
20 [pointed] וּרְחִימוּ וּרְחִימוּ וּדְחִילוּ, לְיַחֲדָא אוֹת יוֹ”ד וְאוֹת הֵ”א
21 [pointed] בְּאוֹת וָא”ו וְאוֹת הֵ”א בְּיִחוּדָא שְׁלִים בְּשֵׁם כָּל־יִשְׂרָאֵל,
22 [pointed] לְדַעַת רִבִּי שִׁמְעוֹן בֶּן יוֹחַאי וְרִבִּי אֶלְעָזָר בְּנוֹ, וּלְדַעַת
23 [pointed] רַבֵּנוּ יִצְחָק לוּרְיָא אַשְׁכְּנַזִּי, הַמָּקוֹם בְּרַחֲמָיו יַעְזְרֵנוּ
