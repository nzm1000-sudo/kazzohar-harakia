01 [pointed] בְּשִׁירוֹת וְתִשְׁבָּחוֹת, שֶׁתֵּפֶן בְּרַחֲמִים אֶל קְרִיאַת מִזְמוֹרֵי תְּהִלִּים
02 [pointed] שֶׁאֶקְרָא כְּאִלּוּ אֲמָרָם דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם בְּעַצְמוֹ זְכוּתוֹ
03 [pointed] יָגֵן עָלֵינוּ. וְיַעֲמָד לָנוּ זְכוּת פְּסוּקֵי תְּהִלִּים וּזְכוּת תֵּבוֹתֵיהֶם
04 [pointed] וְאוֹתִיוֹתֵיהֶם וּנְקֻדּוֹתֵיהֶם וְטַעֲמֵיהֶם וְהַשֵּׁמוֹת הַיּוֹצְאִים מֵהֶם
05 [pointed] מֵרָאשֵׁי תֵּבוֹת וּמִסּוֹפֵי תֵּבוֹת, לְכַפֵּר פְּשָּׁעֵינוּ וַעֲוֹנוֹתֵינוּ וְחַטֹּאתֵינוּ,
06 [pointed] וּלְזַמֵּר עָרִיצִים וּלְהַכְרִית כָּל הַחוֹחִים וְהַקּוֹצִים הַסּוֹבְבִים אֶת
07 [pointed] הַשּׁוֹשַׁנָּה הָעֶלְיוֹנָה, וּלְחַבֵּר אֵשֶׁת נְעוּרִים עִם דּוֹדָהּ בְּאַהֲבָה
08 [pointed] וְאַחֲוָה וְרֵעוּת, וּמִשָּׁם יִמָּשֵׁךְ לָנוּ שֶׁפַע לְנֶפֶשׁ רוּחַ וּנְשָׁמָה לְטַהֲרֵנוּ
09 [pointed] מֵעֲוֹנוֹתֵינוּ וְלִסְלֹחַ חַטֹּאתֵינוּ וּלְכַפֵּר פְּשָּׁעֵינוּ, כְּמוֹ שֶׁסָלַחְתָּ
10 [pointed] לְדָוִד שֶׁאָמַר מִזְמוֹרִים אֵלּוּ לְפָנֶיךָ, כְּמוֹ שֶׁנֶּאֱמַר גַּם יְיָ הֶעֱבִיר
11 [pointed] חַטָּאתְךָ לֹא תָּמוּת. וְאַל תִּקָּחֵנוּ מֵהָעוֹלָם הַזֶּה קֹדֶם זְמַנֵּנוּ עַד
12 [pointed] מְלֹאת שְנוֹתֵינוּ בָּהֶם שִׁבְעִים שָׁנָה בְּאֹפֶן שֶׁנּוּכַל לְתַקֵּן אֵת
13 [pointed] אָשֵׁר שִׁחַתְנוּ. וּזְכוּת דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם יָגֵן עָלֵינוּ וּבַּעֲדֵנוּ
14 [pointed] שְׁתַּאֲרִיךְ אַפְּךָ עַד שׁוּבֵנוּ אֵלֶיךָ בִּתְשׁוּבָה שְׁלֵמָה לְפָנֶיךָ. וּמֵאוֹצַר
15 [pointed] מַתְּנַת חִנָּם חֲנָנוֹ, כְּדִּכְתִיב וְחַנּוֹתִי אֶת אֲשֶׁר אָחֹן וְרִחַמְתִּי אֶת
16 [pointed] אָשֵׁר אֲרַחֵם. וּכְשֵׁם שֶׁאָנוּ אוֹמְרִים לְפָנֶיךָ שִׁירָה בָּעוֹלָם הַזֶּה
17 [pointed] כָּךְ נִזְכֶּה לוֹמַר לְפָנֶיךָ יְיָ אֱלֹהֵינוּ שִׁיר וּשְׁבָחָה לָעוֹלָם הַבָּא. וְעַל
18 [pointed] יְדֵי אֲמִירַת תְּהִלִּים תִּתְעוֹרֵר חֲבַצֶּלֶת הַשָּׁרוֹן וְלָשִׁיר בְּקוֹל נָעִים
19 [pointed] בְּגִילַת וְרַנֵּן, כְּבוֹד הַלְּבָנוֹן נִתָּן לָהּ, הוֹד וְהָדָר בְּבֵית אֱלֹהֵינוּ
20 [pointed] בִּמְהֵרָה בְּיָמֵינוּ אָמֵן סֶלָה:
21 [pointed] לְכוּ נְרַנְּנָה לַיהֹוָה, נָרִיעָה לְצוּר יִשְׁעֵנוּ: נְקַדְּמָה פָּנָיו בְּתוֹדָה,
22 [pointed] בִּזְמִרוֹת נָרִיעַ לוֹ: כִּי אֵל גָּדוֹל יְהֹוָה, וּמֶלֶךְ גָּדוֹל עַל כָּל אֱלֹהִים:
23 [prose] טוב לומר זאת לפני אמירת העשרה מזמורים
24 [pointed] הֲרֵינִי מְקַשֵׁר עַצְמִי בַּאֲמִירַת הָעֲשָׂרָה מִזְמוֹרִים אֵלּוּ לְכָל
25 [pointed] הַצַּדִּיקִים הָאֲמִתִּיִּים שֶׁבְּדוֹרֵנוּ. וּלְכָל הַצַּדִּיקִים הָאֲמִתִּים
