01 [pointed] עַל כֵּן אַפָרֵשׁ לְךָ אֵיךְ תִּתְנַהֵג בְּמִדַּת הָעֲנָוָה, לָלֶכֶת בָּהּ תָּמִיד:
02 [pointed] כָּל דְבָרֶיךָ יִהְיוּ בְּנַחַת, וְרֹאשְׁךָ כָּפוּף; וְעֵינֶךָ יַבִּיטוּ לְמַטָּה לָאָרֶץ,
03 [pointed] וְלִבְּךָ לְמַעֲלָה; וְאַל תַּבִּיט בִּפְנֵי אָדָם בְּדַבֶּרְךָ עִמוֹ. וְכָל אָדָם יִהְיֶה
04 [pointed] גָדוֹל מִמְךָ בְּעֵינֶיךָ: אִם חָכָם אוֹ עָשִׁיר הוּא – עָלֶיךָ לְכַבְּדוֹ. וְאִם
05 [pointed] רָשׁ הוּא, וְאַתָּה עָשִׁיר אוֹ חָכָם מִמֶנוּ – חֲשֹׁב בְּלִבְּךָ כִּי אַתָּה חַיָּב
06 [pointed] מִמֶנוּ, וְהוּא זַכַּאי מִמְךָ, שֶׁאִם הוּא חוֹטֵא – הוּא שׁוֹגֵג, וְאַתָּה מֵזִיד.
07 [pointed] בְּכָל דְּבָרֶיךָ וּמַעֲשֶׂיךָ וּמַחְשְׁבוֹתֶיךָ, וּבְכָל עֵת – חֲשׁוֹב בְּלִבָּךְ כְּאִלוּ
08 [pointed] אַתָּה עוֹמֵד לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא וּשְׁכִינָתוֹ עָלֶיךָ, כִּי כְּבוֹדוֹ
09 [pointed] מָלֵא כָּל הָעוֹלָם. וּדְבָרֶיךָ יִהְיוּ בְּאֵימָה וּבְיִרְאָה, כְּעֶבֶד לִפְנֵי רַבּוֹ.
10 [pointed] וְתִתְבַּיֵּשׁ מִכָּל אָדָם. וְאִם יִקְרָאֲךָ אִישׁ – אַל תַּעֲנֵהוּ בְּקוֹל רָם,
11 [pointed] רַק בְּנַחַת כְּעוֹמֵד לִפְנֵי רַבּוֹ.
12 [pointed] וֶהֱוֵי זָהִיר לִקְרוֹת בַּתּוֹרָה תָּמִיד, אֲשֶׁר תּוּכַל לְקַיְּמָהּ. וְכַאֲשֶׁר
13 [pointed] תָּקוּם מִן הַסֵּפֶר – תְּחַפֵּשֹ בַּאֲשֶׁר לָמַדְתָּ אִם יֵשׁ בּוֹ דָבָר אֲשֶׁר
14 [pointed] תּוּכַל לְקַיְּמוֹ. וּתְפַשְׁפֵּשׁ בְּמַעֲשֶׂיךָ בַּבֹּקֶר וּבָעֶרֶב, וּבָזֶה יִהְיוּ כָּל
15 [pointed] יָמֶיךָ בִּתְשׁוּבָה.
16 [pointed] וְהַסֵר כָּל דִבְרֵי הָעוֹלָם מִלִבְּךָ בְּעֵת הַתְּפִלָּה, וְהָכֵן לִבְּךָ לִפְנֵי
17 [pointed] הַמָּקוֹם בָּרוּךְ הוּא. וְטַהֵר רַעֲיוֹנֶיךָ, וַחֲשב הַדִּבּוּר קדֶם שֶׁתּוֹצִיאֶנּוּ
18 [pointed] מִפִּיךָ.
19 [pointed] וְכֵן תַּעֲשֶׂה כָּל יְמֵי חַיֵּי הֶבְלֶךָ בְּכָל דָּבָר וְדָבָר, וְלא תֶחֱטָא. וּבָזֶּה
20 [pointed] יִהְיוּ דְּבָרֶיךָ וּמַעֲשֶׂיךָ וּמַחְשְׁבוֹתֶיךָ יְשָׁרִים; וּתְפִלָּתְךָ תִּהְיֶה זַכָּה
21 [pointed] וּבָרָה וּנְקִיָּה, וּמְכֻוֶּנֶת וּמְקֻבֶּלֶת לִפְנֵי הַמָּקוֹם בָּרוּךְ הוּא, שֶׁנֶּאֱמַר
22 [pointed] (תהלים י יז): “תָּכִין לִבָּם – תַּקְשִׁיב אָזְנֶךָ”.
23 [pointed] תִּקְרָא הָאִגֶּרֶת הַזאת פַּעַם אַחַת בְּשָׁבוּעַ וְלא תִפְחת, לְקַיְּמָהּ
24 [pointed] וְלָלֶכֶת בָּהּ תָּמִיד אַחַר הַשֵּׁם יִתְבָּרַךְ, לְמַעַן תַּצְלִיחַ בְּכָל דְּרָכֶיךָ,
25 [pointed] וְתִזְכֶּה לָעוֹלָם הַבָּא הַצָּפוּן לַצַּדִּיקִים. וּבְכָל יוֹם שֶׁתִּקְרָאֶנָּה –
