01 [bold] פְשָׁעַי לַיהוָה וְאַתָּה נָשָׂאתָ עֲוֹן חַטָּאתִי סֶלָה: {ו} עַל
02 [bold] זֹאת יִתְפַּלֵּל כָּל חָסִיד אֵלֶיךָ לְעֵת מְצֹא רַק לְשֵׁטֶף מַיִם
03 [bold] רַבִּים אֵלָיו לֹא יַגִּיעוּ: {ז} אַתָּה סֵתֶר לִי | מִצַּר תִּצְּרֵנִי
04 [bold] רָנֵּי פַלֵּט תְּסוֹבְבֵנִי סֶלָה: {ח} אַשְׂכִּילְךָ וְאוֹרְךָ בְּדֶרֶךְ
05 [bold] זוּ תֵלֵךְ אִיעֲצָה עָלֶיךָ עֵינִי: {ט} אַל תִּהְיוּ כְּסוּס כְּפֶרֶד
06 [bold] אֵין הָבִין בְּמֶתֶג וָרֶסֶן עֶדְיוֹ לִבְלוֹם בַּל קְרֹב אֵלֶיךָ:
07 [bold] {י} רַבִּים מַכְאוֹבִים לָרָשָׁע וְהַבּוֹטֵחַ בַּיהוָה חֶסֶד יְסוֹבְבֶנּוּ:
08 [bold] {יא} שִׂמְחוּ בַיהוָה וְגִילוּ צַדִּיקִים וְהַרְנִינוּ כָּל יִשְׁרֵי לֵב:
09 [bold] תהילים פרק-מא
10 [bold] {א} לַמְנַצֵּחַ מִזְמוֹר לְדָוִד: {ב} אַשְׁרֵי מַשְׂכִּיל אֶל דָּל
11 [bold] בְּיוֹם רָעָה יְמַלְּטֵהוּ יְהוָה: {ג} יְהוָה יִשְׁמְרֵהוּ וִיחַיֵּהוּ
12 [bold] (יאשר) וְאֻשַּׁר בָּאָרֶץ וְאַל תִּתְּנֵהוּ בְּנֶפֶשׁ אֹיְבָיו: {ד} יְהוָה
13 [bold] יִסְעָדֶנּוּ עַל עֶרֶשׂ דְּוָי כָּל מִשְׁכָּבוֹ הָפַכְתָּ בְחָלְיוֹ: {ה} אֲנִי
14 [bold] אָמַרְתִּי יְהוָה חָנֵּנִי רְפָאָה נַפְשִׁי כִּי חָטָאתִי לָךְ: {ו} אוֹיְבַי
15 [bold] יֹאמְרוּ רַע לִי מָתַי יָמוּת וְאָבַד שְׁמוֹ: {ז} וְאִם בָּא לִרְאוֹת
16 [bold] שָׁוְא יְדַבֵּר לִבּוֹ יִקְבָּץ אָוֶן לוֹ יֵצֵא לַחוּץ יְדַבֵּר: {ח} יַחַד
17 [bold] עָלַי יִתְלַחֲשׁוּ כָּל שֹׂנְאָי עָלַי יַחְשְׁבוּ רָעָה לִי: {ט} דְּבַר
18 [bold] בְּלִיַּעַל יָצוּק בּוֹ וַאֲשֶׁר שָׁכַב לֹא יוֹסִיף לָקוּם: {י} גַּם אִישׁ
19 [bold] שְׁלוֹמִי אֲשֶׁר בָּטַחְתִּי בוֹ אוֹכֵל לַחְמִי הִגְדִּיל עָלַי עָקֵב:
20 [bold] {יא} וְאַתָּה יְהוָה חָנֵּנִי וַהֲקִימֵנִי וַאֲשַׁלְּמָה לָהֶם: {יב} בְּזֹאת
21 [bold] יָדַעְתִּי כִּי חָפַצְתָּ בִּי כִּי לֹא יָרִיעַ אֹיְבִי עָלָי: {יג} וַאֲנִי
22 [bold] בְּתֻמִּי תָּמַכְתָּ בִּי וַתַּצִּיבֵנִי לְפָנֶיךָ לְעוֹלָם: {יד} בָּרוּךְ
