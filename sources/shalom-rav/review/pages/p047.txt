01 [pointed] יִתֵּן יְיָ יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם.
02 [pointed] יִהְיוּ לְרָצון אִמְרֵי פִי וְהֶגְיון לִבִּי לְפָנֶיךָ, יְיָ צוּרִי וְגאֲלִי. יְיָ צְבָאוֹת
03 [pointed] עִמָּנוּ מִשְׂגָּב לָנוּ אֱלֹהי יַעֲקֹב סֶלָה, יְיָ צְבָאוֹת אַשְׁרֵי אָדָם בֹּטֵחַ
04 [pointed] בָּךְ, יְיָ הוֹשִׁיעָה הַמֶּלֶךְ יַעֲנֵנוּ בְיוֹם קָרְאֵנוּ.
05 [heading] תפילה לשלום חיילי ישראל
06 [pointed] לַמְנַצֵּחַ מִזְמוֹר לְדָוִד: יַעַנְךָ יְיָ בְּיוֹם צָרָה יְשַׂגֶּבְךָ שֵׁם אֱלֹהֵי
07 [pointed] יַעֲקֹב: יִשְׁלַח עֶזְרְךָ מִקֹּדֶשׁ וּמִצִּיּוֹן יִסְעָדֶךָּ: יִזְכֹּר כָּל מִנְחֹתֶךָ
08 [pointed] וְעוֹלָתְךָ יְדַשְּׁנֶה סֶלָה: יִתֶּן לְךָ כִלְבָבֶךָ וְכָל עֲצָתְךָ יְמַלֵּא: נְרַנְּנָה
09 [pointed] בִּישׁוּעָתֶךָ וּבְשֵׁם אֱלֹהֵינוּ נִדְגֹּל יְמַלֵּא יְיָ כָּל מִשְׁאֲלוֹתֶיךָ: עַתָּה
10 [pointed] יָדַעְתִּי כִּי הוֹשִׁיעַ יְיָ מְשִׁיחוֹ יַעֲנֵהוּ מִשְּׁמֵי קָדְשׁוֹ בִּגְבֻרוֹת יֵשַׁע
11 [pointed] יְמִינוֹ: אֵלֶּה בָרֶכֶב וְאֵלֶּה בַסּוּסִים וַאֲנַחְנוּ בְּשֵׁם יְיָ אֱלֹהֵינוּ נַזְכִּיר:
12 [pointed] הֵמָּה כָּרְעוּ וְנָפָלוּ וַאֲנַחְנוּ קַּמְנוּ וַנִּתְעוֹדָד: יְיָ הוֹשִׁיעָה הַמֶּלֶךְ
13 [pointed] יַעֲנֵנוּ בְיוֹם קָרְאֵנוּ:
14 [pointed] יֹשֵׁב בְּסֵתֶר עֶלְיוֹן בְּצֵל שַׁדַּי יִתְלוֹנָן: אֹמַר לַיְיָ מַחְסִי וּמְצוּדָתִי
15 [pointed] אֱלֹהַי אֶבְטַח בּוֹ: כִּי הוּא יַצִּילְךָ מִפַּח יָקוּשׁ מִדֶּבֶר הַוּוֹת: בְּאֶבְרָתוֹ
16 [pointed] יָסֶךְ לָךְ וְתַחַת כְּנָפָיו תֶּחְסֶה צִנָּה וְסֹחֵרָה אֲמִתּוֹ: לֹא תִירָא מִפַּחַד
17 [pointed] לָיְלָה מֵחֵץ יָעוּף יוֹמָם: מִדֶּבֶר בָּאֹפֶל יַהֲלֹךְ מִקֶּטֶב יָשׁוּד צָהֳרָיִם:
18 [pointed] יִפֹּל מִצִּדְּךָ אֶלֶף וּרְבָבָה מִימִינֶךָ אֵלֶיךָ לֹא יִגָּשׁ: רַק בְּעֵינֶיךָ תַבִּיט
19 [pointed] וְשִׁלֻּמַת רְשָׁעִים תִּרְאֶה: כִּי אַתָּה יְיָ מַחְסִי עֶלְיוֹן שַׂמְתָּ מְעוֹנֶךָ:
20 [pointed] לֹא תְאֻנֶּה אֵלֶיךָ רָעָה וְנֶגַע לֹא יִקְרַב בְּאָהֳלֶךָ: כִּי מַלְאָכָיו יְצַוֶּה
21 [pointed] לָּךְ לִשְׁמָרְךָ בְּכָל דְּרָכֶיךָ: עַל כַּפַּיִם יִשָּׂאוּנְךָ פֶּן תִּגֹּף בָּאֶבֶן רַגְלֶךָ:
22 [pointed] עַל שַׁחַל וָפֶתֶן תִּדְרֹךְ תִּרְמֹס כְּפִיר וְתַנִּין: כִּי בִי חָשַׁק וַאֲפַלְּטֵהוּ
23 [pointed] אֲשַׂגְּבֵהוּ כִּי יָדַע שְׁמִי: יִקְרָאֵנִי וְאֶעֱנֵהוּ עִמּוֹ אָנֹכִי בְצָרָה אֲחַלְּצֵהוּ
24 [pointed] וַאֲכַבְּדֵהוּ: אֹרֶךְ יָמִים אַשְׂבִּיעֵהוּ וְאַרְאֵהוּ בִּישׁוּעָתִי:
