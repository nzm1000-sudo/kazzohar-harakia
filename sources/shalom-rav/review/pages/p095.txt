01 [pointed] שְׁרוּיִים לָכֶם, שְׁרוּיִים לָכֶם. מְחוּלִים לָכֶם, מְחוּלִים
02 [pointed] לָכֶם, מְחוּלִים לָכֶם. אֵין כָּאן קְלָלוֹת. אֵין כָּאן חֲרָמִים.
03 [pointed] אֵין כָּאן נִדּוּיִים. אֵין כָּאן נְזִיפוֹת. אֵין כָּאן בִּטּוּיִים. אֵין
04 [pointed] כָּאן דְּבָרִים רָעִים. אֵין כָּאן חֲלוֹמוֹת רָעִים. אֵין כָּאן
05 [pointed] פִּתְרוֹנִים רָעִים. אֵין כָּאן מְסִירַת דִּין. אֵין כָּאן פִּתְחוֹן
06 [pointed] פֶּה לְרָעָה. אֵין כָּאן מַחֲשָׁבוֹת זָרוֹת וְהִרְהוּרִים רָעִים. אֵין
07 [pointed] כָּאן גְּזֵרוֹת קָשׁוֹת וְרָעוֹת. אֵין כָּאן עַיִן הָרַע דְּנָשִׁים. אֵין
08 [pointed] כָּאן עַיִן הָרַע שֶׁל אֲנָשִׁים. אֵין כָּאן עַיִן הָרַע דְּשׂוֹנְאִים
09 [pointed] וּדְאוֹהֲבִים. אֵין כָּאן דְּבָרִים אֲסוּרִים. אֵין כָּאן קַבָּלוֹת.
10 [pointed] אֵין כָּאן מִנְהָגִים. כֻּלָּם יִהְיוּ בְטֵלִים וּמְבֻטָּלִים, וַחֲשׁוּבִים
11 [pointed] כַּחֶרֶס הַנִּשְׁבָּר, וּכְדָבָר שֶׁאֵין בּוֹ מַמָּשׁ, שְׁבִיתִין וּשְׁבִיקִין,
12 [pointed] לָא שְׁרִירִין וְלָא קַיָּמִין. וְכָל־ מִינֵי עַיִן הָרַע יוּסְרוּ מֵעַל
13 [pointed] כָּל יִשְׂרָאֵל וּמֵעֲלֵיכֶם וּמִבָּֽתֵּיכֶם, וְיֻשְׁלְכוּ בִמְצוּלוֹת
14 [pointed] יָם, כַּכָּתוּב: וְלֹא יִדְבַּק בְּיָֽדְךָ מְאוּמָה מִן הַחֵרֶם, לְמַעַן
15 [pointed] יָשׁוּב יְיָ מֵחֲרוֹן אַפּוֹ, וְנָֽתַן לְךָ רַחֲמִים וְרִחַמְךָ וְהִרְבֶּךָ,
16 [pointed] כַּאֲשֶׁר נִשְׁבַּע לַאֲבֹתֶיךָ:
17 [pointed] אָֽנָּֽא יְיָ, הִתְמַלֵּא בְרַחֲמִים, וְיִכְבְּשׁוּ רַחֲמֶיךָ אֶת כַּעֲסֶךָ,
18 [pointed] וְיִגֹּלּוּ רַחֲמֶיךָ עַל מִדּוֹתֶיךָ, וְיִתְמַתְּקוּ הַדִּינִין מֵעַל עַמְּךָ
19 [pointed] יִשְׂרָאֵל וּמֵעָלֵינוּ. וּזְכֹר לָנוּ עֲקֵדָתוֹ שֶׁל יִצְחָק אָבִינוּ
20 [pointed] עָלָיו הַשָּׁלוֹם כְּאִלּוּ אֶפְרוֹ צָבוּר וּמֻנָּח עַל גַּבֵּי הַמִּזְבֵּחַ,
21 [pointed] וְהַבֵּט בְּאֶפְרוֹ לְהַצִּילֵנוּ לָנוּ וּלְכָל יִשְׂרָאֵל אַחֵינוּ מִכָּל
22 [pointed] רָע. וּכְשֵׁם שֶׁמַּסְכִּימִים וּמַתִּירִים לָכֶם בֵּית דִּין שֶׁל
23 [pointed] מַטָּה, כָּךְ יַסְכִּימוּ וְיַתִּירוּ לָכֶם בֵּית דִּין שֶׁל מַעְלָה. וְכָל
