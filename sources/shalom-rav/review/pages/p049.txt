01 [pointed] רוּחִי וְאַתָּה יָדַעְתָּ נְתִיבָתִי בְּאֹרַח זוּ אֲהַלֵּךְ טָמְנוּ פַח לִי: הַבֵּיט
02 [pointed] יָמִין וּרְאֵה וְאֵין לִי מַכִּיר אָבַד מָנוֹס מִמֶּנִּי אֵין דּוֹרֵשׁ לְנַפְשִׁי:
03 [pointed] זָעַקְתִּי אֵלֶיךָ יְיָ אָמַרְתִּי אַתָּה מַחְסִי חֶלְקִי בְּאֶרֶץ הַחַיִּים: הַקְשִׁיבָה
04 [pointed] אֶל רִנָּתִי כִּי דַלּוֹתִי מְאֹד הַצִּילֵנִי מֵרֹדְפַי כִּי אָמְצוּ מִמֶּנִּי: הוֹצִיאָה
05 [pointed] מִמַּסְגֵּר נַפְשִׁי לְהוֹדוֹת אֶת שְׁמֶךָ בִּי יַכְתִּרוּ צַדִּיקִים כִּי תִגְמֹל עָלָי:
06 [pointed] מִי אֵל כָּמוֹךָ נֹשֵׂא עָוֹן וְעֹבֵר עַל פֶּשַׁע לִשְׁאֵרִית נַחֲלָתוֹ לֹא
07 [pointed] הֶחֱזִיק לָעַד אַפּוֹ כִּי חָפֵץ חֶסֶד הוּא: יָשׁוּב יְרַחֲמֵנוּ יִכְבֹּשׁ עֲוֹנֹתֵינוּ
08 [pointed] וְתַשְׁלִיךְ בִּמְצֻלוֹת יָם כָּל חַטֹּאותָם: תִּתֵּן אֱמֶת לְיַעֲקֹב חֶסֶד
09 [pointed] לְאַבְרָהָם אֲשֶׁר נִשְׁבַּעְתָּ לַאֲבֹתֵינוּ מִימֵי קֶדֶם:
10 [pointed] וַיִּסָּעוּ וַיְהִי חִתַּת אֱלֹהִים עַל הֶעָרִים אֲשֶׁר סְבִיבֹתֵיהֶם וְלֹא רָדְפוּ
11 [pointed] אַחֲרֵי בְּנֵי יַעֲקֹב:
12 [pointed] אנא יְיָ אֵל רַחוּם וְחַנּוּן אֶרֶךְ אַפַּיִם וְרַב חֶסֶד וֶאֱמֶת: זְכֹר לְאַבְרָהָם
13 [pointed] לְיִצְחָק וּלְיִשְׂרָאֵל עֲבָדֶיךָ אֲשֶׁר נִשְׁבַּעְתָּ לָהֶם בָּךְ וַתְּדַבֵּר אֲלֵהֶם
14 [pointed] אַרְבֶּה אֶת זַרְעֲכֶם כְּכוֹכְבֵי הַשָּׁמָיִם וְכָל הָאָרֶץ הַזֹּאת אֲשֶׁר אָמַרְתִּי
15 [pointed] אֶתֵּן לְזַרְעֲכֶם וְנָחֲלוּ לְעֹלָם וכתיב: וְזָכַרְתִּי אֶת בְּרִיתִי יַעֲקוֹב וְאַף
16 [pointed] אֶת בְּרִיתִי יִצְחָק וְאַף אֶת בְּרִיתִי אַבְרָהָם אֶזְכֹּר וְהָאָרֶץ אֶזְכֹּר:
17 [pointed] ועשה למען רחמך וחסדך ובזכות פרקי תהילים והפסוקים
18 [pointed] אשר קראתי, תשמור נא יְיָ אלוהינו מעתה ועד עולם את
19 [pointed] כל חיילי ישראל, בכל מקום שהם מכול צר ואויב ומזיק
20 [pointed] ומכול מקרים רעים, וכול היועץ עליהם ועל עמך בית ישראל
21 [pointed] עצה שאינה טובה ומחשבה שאינה טובה הָפֵר עֲצָתָם וקלקל
22 [pointed] מחשבותם: יִהְיוּ כְּמוֹץ לִפְנֵי רוּחַ וּמַלְאַךְ יְיָ דּוֹחֶה, תִּפֹּל עֲלֵיהֶם
23 [pointed] אֵימָתָה וָפַחַד בִּגְדֹל זְרוֹעֲךָ יִדְּמוּ כָּאָבֶן. ובעבור שמך אֵל שַׁדַּי
24 [pointed] בטל מעליהם כל משׂטנים ומקטרגים והצילם משעות קשות
25 [pointed] ורעות המתרגשות ובאות לעולם. וקיים בהם: לֹא יִתְיַצֵּב אִישׁ
26 [pointed] בִּפְנֵיכֶם פַּחְדְּכֶם וּמוֹרַאֲכֶם יִתֵּן יְיָ אֱלֹהֵיכֶם עַל פְּנֵי כָל הָאָרֶץ
