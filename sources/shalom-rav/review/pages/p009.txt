01 [pointed] אֲנִי ה’ אֱלהֵיכֶם אֲשֶׁר הוצֵאתִי אֶתְכֶם מֵאֶרֶץ מִצְרַיִם לִהְיות לָכֶם
02 [pointed] לֵאלהִים. אֲנִי ה’ אֱלהֵיכֶם:
03 [pointed] וַיְדַבֵּר יְהוָה אֶל משֶׁה לֵּאמר: קַדֶּשׁ לִי כָל בְּכוֹר פֶּטֶר כָּל רֶחֶם
04 [pointed] בִּבְנֵי יִשְׂרָאֵל בָּאָדָם וּבַבְּהֵמָה לִי הוּא: וַיּאמֶר משֶׁה אֶל הָעָם
05 [pointed] זָכוֹר אֶת הַיּוֹם הַזֶּה אֲשֶׁר יְצָאתֶם מִמִּצְרַיִם מִבֵּית עֲבָדִים כִּי
06 [pointed] בְּחזֶק יָד הוֹצִיא יְהוָֹה אֶתְכֶם מִזֶּה וְלא יֵאָכֵל חָמֵץ: הַיּוֹם אַתֶּם
07 [pointed] יצְאִים בְּחדֶשׁ הָאָבִיב: וְהָיָה כִי יְבִיאֲךָ יְהוָה אֶל אֶרֶץ הַכְּנַעֲנִי
08 [pointed] וְהַחִתִּי וְהָאֱמרִי וְהַחִוִּי וְהַיְבוּסִי אֲשֶׁר נִשְׁבַּע לַאֲבתֶיךָ לָתֶת לָךְ
09 [pointed] אֶרֶץ זָבַת חָלָב וּדְבָשׁ וְעָבַדְתָּ אֶת הָעֲבדָה הַזּאת בַּחדֶשׁ הַזֶּה:
10 [pointed] שִׁבְעַת יָמִים תּאכַל מַצּת וּבַיּוֹם הַשְּׁבִיעִי חַג לַיהוָה: מַצּוֹת
11 [pointed] יֵאָכֵל אֵת שִׁבְעַת הַיָּמִים וְלא יֵרָאֶה לְךָ חָמֵץ וְלא יֵרָאֶה לְךָ שְׂאר
12 [pointed] בְּכָל גְּבֻלֶךָ: וְהִגַּדְתָּ לְבִנְךָ בַּיּוֹם הַהוּא לֵאמר בַּעֲבוּר זֶה עָשָׂה
13 [pointed] יְהוָה לִי בְּצֵאתִי מִמִּצְרָיִם: וְהָיָה לְךָ לְאוֹת עַל יָדְךָ וּלְזִכָּרוֹן בֵּין
14 [pointed] עֵינֶיךָ לְמַעַן תִּהְיֶה תּוֹרַת יְהוָה בְּפִיךָ כִּי בְּיָד חֲזָקָה הוֹצִאֲךָ יְהוָֹה
15 [pointed] מִמִּצְרָיִם: וְשָׁמַרְתָּ אֶת הַחֻקָּה הַזּאת לְמוֹעֲדָהּ מִיָּמִים יָמִימָה:
16 [pointed] וְהָיָה כִּי יְבִאֲךָ יְהוָה אֶל אֶרֶץ הַכְּנַעֲנִי כַּאֲשֶׁר נִשְׁבַּע לְךָ וְלַאֲבתֶיךָ
17 [pointed] וּנְתָנָהּ לָךְ: וְהַעֲבַרְתָּ כָל פֶּטֶר רֶחֶם לַיהוָֹה וְכָל פֶּטֶר שֶׁגֶר בְּהֵמָה
18 [pointed] אֲשֶׁר יִהְיֶה לְךָ הַזְּכָרִים לַיהוָה: וְכָל פֶּטֶר חֲמר תִּפְדֶּה בְשֶׂה וְאִם לא
19 [pointed] תִפְדֶּה וַעֲרַפְתּוֹ וְכל בְּכוֹר אָדָם בְּבָנֶיךָ תִּפְדֶּה: וְהָיָה כִּי יִשְׁאָלְךָ בִנְךָ
20 [pointed] מָחָר לֵאמר מַה זּאת וְאָמַרְתָּ אֵלָיו בְּחזֶק יָד הוֹצִיאָנוּ יְהוָה מִמִּצְרַיִם
21 [pointed] מִבֵּית עֲבָדִים: וַיְהִי כִּי הִקְשָׁה פַרְעה לְשַׁלְּחֵנוּ וַיַּהֲרג יְהוָֹה כָּל
22 [pointed] בְּכוֹר בְּאֶרֶץ מִצְרַיִם מִבְּכר אָדָם וְעַד בְּכוֹר בְּהֵמָה עַל כֵּן אֲנִי זבֵחַ
23 [pointed] לַיהוָה כָּל פֶּטֶר רֶחֶם הַזְּכָרִים וְכָל בְּכוֹר בָּנַי אֶפְדֶּה: וְהָיָה לְאוֹת
24 [pointed] עַל יָדְכָה וּלְטוֹטָפת בֵּין עֵינֶיךָ כִּי בְּחזֶק יָד הוֹצִיאָנו יְּהוָה מִמִּצְרָיִם:
