01 [pointed] (יכוון בשם: תעאו). וְכָל צִדְקוֹת יִשְרָאֵל וְתַלְמוּד תּוֹרָה אֲשֶׁר
02 [pointed] עָשֹוּ וְלָמְדוּ בְּעִירֵנוּ וּבַעֲיָירוֹת אֲשֶׁר בִּגְבוּלֵנוּ יִזָּכְרוּ לִפְנֵי כִסֵּא
03 [pointed] רַחֲמֶיךָ. וּתְבַטֵּל מֵעָלֵינוּ וּמֵעַל יִשְרָאֵל אֲשֶׁר בְּעִיר זוּ (יאמר את
04 [pointed] שם העיר) כָּל גְּזֵירוֹת קָשׁוֹת וְרָעוֹת (יכוון בשם: קרע שטן).
05 [pointed] וְתָחוֹן עָלֵינוּ וַעֲלֵיהֶם מֵאוֹצַר מַתְּנַת חִנָּם כְּכָל צִדְקוֹתֶיךָ. וְתַצִּילֵנוּ
06 [pointed] מִכָּל רָעָב וְיוֹקֶר הַשְּׁעָרִים:
07 [pointed] אֵל מָלֵא רַחֲמִים תֵּן שָׁלוֹם בָּאָרֶץ תֵּן שָׁלוֹם בַּמַּלְכֻיּוֹת וּשְׂרָרוֹת.
08 [pointed] וּבְצֵל כְּנָפֶיךָ תַּסְתִּירֵנוּ בִּזְכוּת אַבְרָהָם אָבִינוּ עָלָיו הַשָּׁלוֹם אִיש
09 [pointed] הַחֶסֶד, וּבִזְכוּת יִצְחָק אָבִינוּ עָלָיו הַשָּׁלוֹם אָזוּר בִּגְבוּרָה, וּבִזְכוּת
10 [pointed] יַעֲקֹב אָבִינוּ עָלָיו הַשָּׁלוֹם כְּלִיל תִּפְאֶרֶת, וּבִזְכוּת משֶׁה רַעְיָא
11 [pointed] מְהֵימְנָא אָחוּז בְּנֵצַח, וּבִזְכוּת אַהֲרֹן כַּהֲנָא רַבָּא אָחוּז בְּהוֹד,
12 [pointed] וּבִזְכוּת יוֹסֵף הַצַּדִּיק אָחוּז בִּיסוֹד, וּבִזְכוּת דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם
13 [pointed] אָחוּז בְּמַלְכוּת, עָזְרֵנוּ אֱלוֹהֵי יִשְׁעֵנוּ עַל דְּבַר כְּבוֹד שְׁמֶךָ וְהַצִּילֵנוּ
14 [pointed] וְכַפֵּר עַל חַטֹּאתֵינוּ לְמַעַן שְׁמֶךָ. רַחוּם חַנּוּן שׁוֹמֵר תּוֹמֵךְ מַצִּיל
15 [pointed] יָשָׁר פּוֹדֶה (יכוון בשם: רחש תמיף). הַצִּילֵנוּ מִכָּל צָרָה וְטִרוֹף
16 [pointed] וּבִלְבּוּל. עֲשֵה לְמַעַן שְׁמֶךָ: עֲשֵה לְמַעַן יְמִינֶךָ: עֲשֵה לְמַעַן תּוֹרָתֶךָ:
17 [pointed] עֲשֵה לְמַעַן קְדֻשָּׁתֶךָ: עֲשֵה לְמַעַן יֵחָלְצוּן יְדִידֶךָ: הוֹשִׁיעָה יְמִינְךָ
18 [pointed] וַעֲנֵנִי. יִהְיוּ לְרָצוֹן אִמְרֵי פִי וְהֶגְיוֹן לִבִּי לְפָנֶיךָ יְיָ צוּרִי וְגֹאֲלִי.
19 [heading] להנצל הוא ובני ביתו מן הסכנה
20 [prose] (יאמר אותו בכל יום פעם אחת - ואם ח”ו מתקיימת התקפה
21 [prose] אווירית יאמר אותו פעם אחר פעם בלי הפסק ובעז”ה לא
22 [prose] יאונה לו כל רע).
23 [pointed] לַמְנַצֵּחַ מִזְמוֹר לְדָוִד: יַעַנְךָ יְיָ בְּיוֹם צָרָה יְשַׂגֶּבְךָ שֵׁם אֱלֹהֵי
24 [pointed] יַעֲקֹב: יִשְׁלַח עֶזְרְךָ מִקֹּדֶשׁ וּמִצִּיּוֹן יִסְעָדֶךָּ: יִזְכֹּר כָּל מִנְחֹתֶךָ
