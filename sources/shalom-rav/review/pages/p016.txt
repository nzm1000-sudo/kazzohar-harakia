01 [bold] לִפְלֵטָה, לְטוֹבָה, לְחֵן, לְחֶסֶד וּלְרַחֲמִים לְחַיִּים טוֹבִים וּלְשָׁלוֹם בְּיוֹם:
02 [bold] (בראש חדש:) רֹאשׁ חֹדֶשׁ הַזֶּה,
03 [bold] (בחוה”מ פסח:) חַג הַמַּצּוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,
04 [bold] (בחוה”מ סוכות:) חַג הַסֻּכּוֹת הַזֶּה, בְּיוֹם מִקְרָא קֹדֶשׁ הַזֶּה,
05 [bold] לְרַחֵם בּוֹ עָלֵינוּ וּלְהוֹשִׁיעֵנוּ. זָכְרֵנוּ יְהֹוָה אֱלֹהֵינוּ בּוֹ לְטוֹבָה, וּפָּקְדֵנוּ
06 [bold] בוֹ לִבְרָכָה, וְהוֹשִׁיעֵנוּ בוֹ לְחַיִּים טוֹבִים, בִּדְבַר יְשׁוּעָה וְרַחֲמִים. חוּס
07 [bold] וְחָנֵּנוּ, וַחֲמֹל וְרַחֵם עָלֵינוּ, וְהוֹשִׁיעֵנוּ כִּי אֵלֶיךָ עֵינֵינוּ, כִּי אֵל מֶלֶךְ
08 [bold] חַנּוּן וְרַחוּם אָתָּה:
09 [bold] וְאַתָּה, בְּרַחֲמֶיךָ הָרַבִּים, תַּחְפֹּץ בָּנוּ וְתִרְצֵנוּ. וְתֶחֱזֶינָה
10 [bold] עֵינֵינוּ בְּשׁוּבְךָ לְצִיּוֹן בְּרַחֲמִים. בָּרוּךְ אַתָּה יִהִוִהִ, הַמַּחֲזִיר
11 [bold] שְׁכִינָתוֹ לְצִיּוֹן:
12 [bold] (כשיאמר מודים ישתחווה)
13 [bold] מוֹדִים אֲנַחְנוּ לָךְ, שָׁאַתָּה הוּא יְהֹוָה אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ
14 [bold] לְעוֹלָם וָעֶד, צוּרֵנוּ צוּר חַיֵּינוּ וּמָגֵן יִשְׁעֵנוּ אַתָּה הוּא לְדֹר
15 [bold] וָדֹר. נוֹדֶה לְּךָ וּנְסַפֵּר תְּהִלָּתֶךָ, עַל חַיֵּינוּ הַמְּסוּרִים בְּיָדֶיךָ,
16 [bold] וְעַל נִשְׁמוֹתֵינוּ הַפְּקוּדוֹת לָךְ, וְעַל נִסֶּיךָ שֶׁבְּכָל יוֹם עִמָּנוּ,
17 [bold] וְעַל נִפְלְאוֹתֶיךָ וְטוֹבוֹתֶיךָ שֶׁבְּכָל עֵת, עֶרֶב וָבֹקֶר וְצָהֳרָיִם.
18 [bold] הַטּוֹב, כִּי לֹא כָלוּ רַחֲמֶיךָ, הַמְּרַחֵם, כִּי לֹא תַמּוּ חֲסָדֶיךָ,
19 [bold] כִּי מֵעוֹלָם קִוִּינוּ לָךְ:
