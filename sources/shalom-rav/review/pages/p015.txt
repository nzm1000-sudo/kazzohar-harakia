01 [bold] שְׁמַע קוֹלֵנוּ יְהֹוָה אֱלֹהֵינוּ, אָב הָרַחֲמָן רַחֵם עָלֵינוּ, וְקַבֵּל
02 [bold] בְּרַחֲמִים וּבְרָצוֹן אֶת תְּפִלָּתֵנוּ, כִּי אֵל שׁוֹמֵעַ תְּפִלּוֹת
03 [bold] וְתַחֲנוּנִים אָתָּה, וּמִלְּפָנֶיךָ מַלְכֵּנוּ, רֵיקָם אַל תְּשִׁיבֵנוּ. חָנֵּנוּ
04 [bold] וַעֲנֵנוּ וּשְׁמַע תְּפִלָּתֵנוּ.
05 [bold] (בתענית יחיד ובתענית ציבור אומר כאן היחיד עננו, וכן שליח
06 [bold] ציבור ששכח יאמר כאן עננו ללא חתימה)
07 [bold] עֲנֵנוּ אָבִינוּ עֲנֵנוּ בַּיּוֹם צוֹם הַתַּעֲנִית הַזֶּה, כִּי בְצָרָה גְּדוֹלָה אֲנַחְנוּ,
08 [bold] אֶל תֵּפֶן לְרִשְעֵינוּ, וְאַל תִּתְעַלַּם מַלְכֵּנוּ מִבַּקָּשָׁתֵנוּ, הָיָה נָא קָרוֹב
09 [bold] לְשַׁוְעָתֵנוּ, טֶרֶם נִקְרָא אֵלֶיךָ אַתָּה תַעֲנֶה, נְדַבֵּר וְאַתָּה תִּשְׁמַע, כַּדָּבָר
10 [bold] שֶׁנֶּאֱמַר: וְהָיָה טֶרֶם יִקְרָאוּ וַאֲנִי אֶעֱנֶה, עוֹד הֵם מְדַבְּרִים וַאֲנִי אֶשְׁמָע:
11 [bold] כִּי אַתָּה יְהֹוָה פּוֹדֶה וּמַצִּיל וְעוֹנֶה וּמְרַחֵם בְּכָל עֵת צָרָה וְצוּקָה:
12 [bold] כִּי אַתָּה שׁוֹמֵעַ תְפִלַּת כָּל פֶּה, בָּרוּךְ אַתָּה יֹהְוָה, שׁוֹמֵעַ
13 [bold] תְּפִלָּה.
14 [bold] רְצֵה יְהֹוָה אֱלֹהֵינוּ בְּעַמְּךָ יִשְׂרָאֵל, וּלִתְפִלָּתָם שְׁעֵה, וְהָשֵׁב
15 [bold] הָעֲבוֹדָה לִדְבִיר בֵּיתֶךָ, וְאִשֵּׁי יִשְׂרָאֵל וּתְפִלָּתָם מְהֵרָה
16 [bold] בְאַהֲבָה תְּקַבֵּל בְּרָצוֹן, וּתְהִי לְרָצוֹן תָּמִיד עֲבוֹדַת יִשְׂרָאֵל
17 [bold] עַמֶּךָ:
18 [bold] (בראש חודש ובחול המועד מוסיפים:)

20 [bold] יִפָּקֵד וְיִזָּכֵר, זִכְרוֹנֵנוּ וְזִכְרוֹן אֲבוֹתֵינוּ, זִכְרוֹן יְרוּשָׁלַיִם עִירָךְ,
