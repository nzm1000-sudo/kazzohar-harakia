01 [pointed] הַקְּלָלוֹת וְכָל הַחֲלוֹמוֹת רָעִים, וְכָל פִּתְרוֹנִים רָעִים,
02 [pointed] וְכָל מְסִירַת דִּין, וְכָל פִּתְחוֹן פֶּה לְרָעָה, וְכָל מִינֵי גְזֵרוֹת
03 [pointed] קָשׁוֹת וְרָעוֹת, וְכָל מִינֵי דְבָרִים אֲסוּרִים, וְכָל מִינֵי
04 [pointed] מַחֲשָׁבוֹת זָרוֹת וְהִרְהוּרִים רָעִים, כֻּלָּם יִתְהַפְּכוּ לָכֶם
05 [pointed] לְטוֹבָה וְלִבְרָכָה, כַּכָּתוּב: וְלֹא אָבָה יְיָ אֱלֹהֶיךָ לִשְׁמוֹעַ
06 [pointed] אֶל בִּלְעָם, וַיַּהֲפֹךְ יְיָ אֱלֹהֶיךָ לְּךָ אֶת הַקְּלָלָה לִבְרָכָה, כִּי
07 [pointed] אֲהֵבְךָ יְיָ אֱלֹהֶיךָ: וּכְתִיב: וְאַתֶּם הַדְּבֵקִים בַּֽייָ אֱלֹהֵיכֶם,
08 [pointed] חַיִּים כֻּלְּכֶם הַיּוֹם: עֲשֵׂה לְמַעַן שְׁמָךְ, עֲשֵׂה לְמַעַן יְמִינָךְ,
09 [pointed] עֲשֵׂה לְמַעַן תּוֹרָתָךְ, עֲשֵׂה לְמַעַן קְדֻשָּׁתָךְ. יִהְיוּ לְרָצוֹן
10 [pointed] אִמְרֵי־פִי, וְהֶגְיוֹן לִבִּי לְפָנֶיךָ. יְיָ צוּרִי וְגֹאֲלִי: יְיָ עֹז לְעַמּוֹ
11 [pointed] יִתֵּן, יְיָ יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם.
12 [prose] ( ואחר כך יאמרו השואלים מודעה זו:)
13 [pointed] הֲרֵי אֲנַחְנוּ מוֹסְרִים מוֹדָעָה לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא,
14 [pointed] וְלִפְנֵי בֵית דִּין שֶׁל מַעְלָה וְלִפְנֵי בֵית דִּין שֶׁל מַטָּה,
15 [pointed] וְלִפְנֵיכֶם, וּמְבַטְּלִים מֵעַתָּה וְעַד עוֹלָם כָּל־מִין קַבָּלָה
16 [pointed] אִם נְקַבֵּל עָלֵינוּ, מִנְּדָרִים וּשְׁבוּעוֹת וּנְזִירוֹת וְאִסּוּרִין
17 [pointed] וְקוּנָמוֹת וַחֲרָמוֹת וְהַסְכָּמוֹת וְקַבָּלָה בַלֵּב, בְּכָל־לָשׁוֹן,
18 [pointed] בְּכָל־צַד וְאֹפֶן שֶׁיִּהְיֶה, בֵּין בְּהָקִיץ בֵּין בַּחֲלוֹם, חוּץ
19 [pointed] מִנִּדְרֵי קַבָּלַת תַּעֲנִית אִם נְקַבֵּל. וּתְנַאי מוֹדָעָה זוֹ יִהְיֶה
20 [pointed] קַיָּם עַד עוֹלָם, וְלֹא יִתְבַּטֵּל בְּשׁוּם צַד וְאֹפֶן שֶׁיִּהְיֶה.
21 [pointed] וְאִם נִשְׁכַּח לִתְנַאי מוֹדָעָה זוֹ, תּוֹךְ כְּדֵי דִבּוּר לְנֶדֶר אוֹ
22 [pointed] לִשְׁבוּעָה, (אוֹ בְשָׁעָה שֶׁאֲנַחְנוּ זוֹכְרִים לִתְנַאי מוֹדָעָה
23 [pointed] זוֹ) וְנַחֲזֹר לִנְדֹּר אוֹ לִשָּׁבַע מֵחֲמַת אֵיזוֹ סִבָּה שֶׁתִּהְיֶה,
